def bubble_sort(arr):     n = len(arr)
    # Outer loop: n-1 passes are enough     for i in range(n - 1):
        # Last i elements are already sorted         for j in range(0, n - i - 1):
            # Compare adjacent elements             if arr[j] > arr[j+1]:
                # Swap using a temp variable
                temp = arr[j]                 arr[j] = arr[j+1]                 arr[j+1] = temp
